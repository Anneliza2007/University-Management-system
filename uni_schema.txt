-- =============================================
-- University Management Database Schema (Oracle SQL)
-- =============================================

CREATE TABLE DEPARTMENT (
    DEPT_ID NUMBER(4) PRIMARY KEY,
    DEPT_NAME VARCHAR2(50) NOT NULL,
    BUILDING VARCHAR2(50),
    BUDGET NUMBER(12, 2)
);

CREATE TABLE TEACHER (
    TEACHER_ID NUMBER(6) PRIMARY KEY,
    FIRST_NAME VARCHAR2(50) NOT NULL,
    LAST_NAME VARCHAR2(50) NOT NULL,
    EMAIL VARCHAR2(100) UNIQUE,
    PHONE VARCHAR2(20),
    DEPARTMENT VARCHAR2(50),
    HIRE_DATE DATE,
    SALARY NUMBER(10, 2)
);

CREATE TABLE ROOM (
    ROOM_ID NUMBER(4) PRIMARY KEY,
    ROOM_NUMBER VARCHAR2(20) NOT NULL,
    BUILDING VARCHAR2(50) NOT NULL,
    ROOM_TYPE VARCHAR2(30) DEFAULT 'Classroom',
    CAPACITY NUMBER(4) NOT NULL
);
